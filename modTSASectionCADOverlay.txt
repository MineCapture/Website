Attribute VB_Name = "modTSASectionCADOverlay"
Option Explicit

Private Const TSA_APP_SUPABASE_URL As String = "https://YOUR-PROJECT.supabase.co"
Private Const TSA_APP_PUBLISHABLE_KEY As String = "YOUR-PUBLISHABLE-KEY"
Private Const TSA_APP_OUTPUT_FOLDER As String = "C:\Temp"

' TSA CAD overlay exporter + Supabase upload - Stage 2.1 (single module).
' Requires JsonConverter.bas (VBA-JSON).
' IMPORTANT: active UCS in the section drawing must map UCS X=chainage, Y=elevation.
' Export entity geometry is converted WCS -> CURRENT UCS, just as GEO surface export.
' No entities are created, exploded, moved or deleted in AutoCAD.

Private Const TSA_OVERLAY_VERSION As Long = 1
Private Const TSA_MAX_BLOCK_DEPTH As Long = 10
Private Const TSA_CIRCLE_MATCH_TOLERANCE As Double = 0.1 ' drawing units
Private mDoc As AcadDocument
Private mWarnings As Collection
Private mCounts As Object
Private mCircles As Collection
Private mHatches As Collection
Private mBlockHandle As String
Private mFallback As Object

' =============================================================
' USER CONFIGURATION: change this function to include/exclude layers.
' =============================================================
Public Function TSA_CADExportLayers() As Variant
    TSA_CADExportLayers = Array("Planning", "XoversXsect", "XoversOffSection")
End Function

Public Function TSA_ShouldExportLayer(ByVal layerName As String) As Boolean
    Dim v As Variant
    For Each v In TSA_CADExportLayers()
        If StrComp(CStr(v), layerName, vbTextCompare) = 0 Then
            TSA_ShouldExportLayer = True
            Exit Function
        End If
    Next v
End Function

' One-line entrypoint from your VBA form or Immediate Window.
' Example: TSA_ExportCADOverlayJSON 8729, "C:\Temp\TSA_section_overlay.json"
Public Sub TSA_ExportCADOverlayJSON(ByVal RdID As Long, ByVal outputPath As String)
    TSA_ExportCADOverlayJSONWithFallback RdID, outputPath, ""
End Sub

Public Sub TSA_ExportCADOverlayJSONWithFallback(ByVal RdID As Long, _
    ByVal outputPath As String, ByVal fallbackPath As String)
    Dim root As Object, section As Object, allLayers As Collection
    Dim ent As AcadEntity, item As Object, layerName As Variant, layerDoc As Object
    Dim ents As Collection, i As Long, exported As Long
    Dim layerCounts As Object, summary As Object, warnArray As Collection
    Dim ff As Integer, json As String, name As String, exists As Boolean
    Dim oldErr As String

    If RdID <= 0 Then Err.Raise vbObjectError + 7001, , "RdID must be positive."
    Set mDoc = ThisDrawing
    TSA_LoadFallback fallbackPath
    mBlockHandle = vbNullString
    Set mWarnings = New Collection
    Set mCounts = CreateObject("Scripting.Dictionary")
    Set root = TSA_Dict()
    Set section = TSA_Dict()
    section.Add "RdID", RdID
    section.Add "coordinateSystem", "CHAINAGE_ELEVATION"
    section.Add "units", "metres"
    root.Add "formatVersion", TSA_OVERLAY_VERSION
    root.Add "source", "CAD_OVERLAY"
    root.Add "sourceDrawing", mDoc.Name
    root.Add "sourceRevision", Format$(Now, "yyyy-mm-dd\Thh:nn:ss")
    root.Add "section", section
    Set allLayers = New Collection

    For Each layerName In TSA_CADExportLayers()
        name = CStr(layerName)
        exists = TSA_LayerExists(name)
        Set ents = New Collection
        Set layerCounts = TSA_Dict()
        Set mCircles = New Collection
        Set mHatches = New Collection
        If Not exists Then
            TSA_Warn name, "", "LAYER", "Configured layer does not exist. Snapshot marked incomplete."
        Else
            For Each ent In mDoc.ModelSpace
                If StrComp(ent.Layer, name, vbTextCompare) = 0 Then
                    On Error GoTo EntityError
                    TSA_ExportCADEntity ent, ents, name, TSA_Identity(), 0
                    On Error GoTo 0
                End If
NextEnt:
            Next ent
            TSA_MatchHatchCircles ents, name
        End If
        Set layerDoc = TSA_Dict()
        layerDoc.Add "layerName", name
        layerDoc.Add "layerExists", exists
        layerDoc.Add "entities", ents
        allLayers.Add layerDoc
        Debug.Print "TSA overlay layer: " & name & " -> " & ents.Count & " primitives"
    Next layerName

    root.Add "layers", allLayers
    Set warnArray = New Collection
    For i = 1 To mWarnings.Count: warnArray.Add mWarnings(i): Next i
    root.Add "warnings", warnArray
    Set summary = TSA_Dict()
    summary.Add "warningCount", mWarnings.Count
    summary.Add "layerCount", allLayers.Count
    root.Add "summary", summary
    json = JsonConverter.ConvertToJson(root, Whitespace:=2)
    ff = FreeFile
    On Error GoTo SaveError
    Open outputPath For Output As #ff
    Print #ff, json
    Close #ff
    Debug.Print "TSA overlay JSON: " & outputPath
    Debug.Print "Warnings: " & mWarnings.Count & " (see JSON warnings array)"
    MsgBox "CAD overlay JSON written." & vbCrLf & _
           "Layers: " & allLayers.Count & vbCrLf & _
           "Warnings: " & mWarnings.Count & vbCrLf & outputPath, vbInformation
    Exit Sub
EntityError:
    oldErr = Err.Description
    TSA_Warn name, ent.Handle, ent.ObjectName, "Error extracting entity: " & oldErr
    Err.Clear
    Resume NextEnt
SaveError:
    On Error Resume Next
    Close #ff
    On Error GoTo 0
    Err.Raise vbObjectError + 7002, "TSA_ExportCADOverlayJSON", "Cannot save file: " & outputPath
End Sub

Private Sub TSA_ExportCADEntity(ByVal ent As AcadEntity, ByRef out As Collection, _
                                ByVal exportLayer As String, ByVal xf As Variant, ByVal depth As Long)
    Dim typ As String, row As Object, geom As Collection, coords As Variant
    Dim c As Variant, p As Variant, q As Variant, col As Object
    Dim i As Long, stride As Long, br As AcadBlockReference
    Dim blockDef As AcadBlock, child As AcadEntity, childXf As Variant
    Dim textPt As Variant, rr As Double, s As String
    Dim previousBlock As String, basePt As Variant
    typ = ent.ObjectName
    If typ = "AcDbBlockReference" Then
        Set br = ent
        If depth >= TSA_MAX_BLOCK_DEPTH Then
            TSA_Warn exportLayer, ent.Handle, typ, "Block recursion limit reached."
            Exit Sub
        End If
        On Error GoTo BlockFail
        Set blockDef = mDoc.Blocks(br.Name)
        If blockDef.IsXRef Then
            TSA_Warn exportLayer, ent.Handle, typ, "XREF block skipped; bind or expand separately."
            Exit Sub
        End If
        childXf = TSA_Compose(xf, TSA_BlockTransform(br))
        'Block definitions can have a non-zero base point.
        basePt = blockDef.Origin
        childXf = TSA_Compose(childXf, TSA_Translation( _
            -CDbl(basePt(0)), -CDbl(basePt(1)), -CDbl(basePt(2))))
        previousBlock = mBlockHandle
        mBlockHandle = br.Handle
        'XCPosition is a section-plane block whose local Y is section elevation.
        'The generic WCS/OCS transform incorrectly makes its line horizontal.
        'Export this known block using the section-space anchor and live attributes.
        If StrComp(CStr(br.EffectiveName), "XCPosition", vbTextCompare) = 0 Then
            TSA_ExportXCPosition br, blockDef, childXf, out, exportLayer
            mBlockHandle = previousBlock
            Exit Sub
        End If
        For Each child In blockDef
            If child.ObjectName <> "AcDbAttributeDefinition" Then
                TSA_ExportCADEntity child, out, exportLayer, childXf, depth + 1
            End If
        Next child
        mBlockHandle = previousBlock
        Exit Sub
BlockFail:
        mBlockHandle = vbNullString
        TSA_Warn exportLayer, ent.Handle, typ, "Block cannot be expanded: " & Err.Description
        Err.Clear
        Exit Sub
    End If

    Set row = TSA_BaseEntity(ent, exportLayer)
    Select Case typ
        Case "AcDbLine"
            row.Add "type", "LINE"
            Set geom = New Collection
            geom.Add TSA_Point(ent.StartPoint, xf)
            geom.Add TSA_Point(ent.EndPoint, xf)
            row.Add "points", geom

        Case "AcDbCircle"
            row.Add "type", "CIRCLE"
            row.Add "centre", TSA_Point(ent.Center, xf)
            rr = CDbl(ent.Radius) * TSA_ApproxScale(xf)
            row.Add "radius", rr
            ' Hatch matcher will set fill for circles with matching solid hatches.
            row.Add "fill", vbNullString
            If TSA_IsBoreholeLayer(exportLayer) Then row("colour") = "#000000"
            mCircles.Add row

        Case "AcDbPolyline", "AcDb2dPolyline", "AcDb3dPolyline"
            row.Add "type", "POLYLINE"
            coords = ent.Coordinates
            If typ = "AcDbPolyline" Then
                stride = 2
            ElseIf typ = "AcDb3dPolyline" Then
                stride = 3
            Else
                ' Legacy 2DPolyline coordinates may be XYZ; ignore elevation if flat.
                stride = 3
            End If
            Set geom = New Collection
            For i = LBound(coords) To UBound(coords) - (stride - 1) Step stride
                If stride = 2 Then
                    p = TSA_XYZ(CDbl(coords(i)), CDbl(coords(i + 1)), TSA_EntityElevation(ent))
                Else
                    p = TSA_XYZ(CDbl(coords(i)), CDbl(coords(i + 1)), CDbl(coords(i + 2)))
                End If
                geom.Add TSA_Point(p, xf)
            Next i
            row.Add "points", geom
            row.Add "closed", TSA_EntityClosed(ent)
            If typ = "AcDbPolyline" Then
                TSA_AppendBulgePoints ent, geom, xf, exportLayer
                Set row("points") = geom
            End If

        Case "AcDbText", "AcDbMText"
            row.Add "type", IIf(typ = "AcDbText", "TEXT", "MTEXT")
            row.Add "text", CStr(ent.TextString)
            If typ = "AcDbText" Then
                textPt = ent.InsertionPoint
                row.Add "height", CDbl(ent.Height) * TSA_ApproxScale(xf)
                row.Add "rotation", TSA_AngleInSection(textPt, CDbl(ent.Rotation), xf)
            Else
                textPt = ent.InsertionPoint
                row.Add "height", CDbl(ent.Height) * TSA_ApproxScale(xf)
                row.Add "rotation", TSA_AngleInSection(textPt, CDbl(ent.Rotation), xf)
                row.Add "width", CDbl(ent.Width) * TSA_ApproxScale(xf)
                row.Add "attachment", CLng(ent.AttachmentPoint)
            End If
            row.Add "position", TSA_Point(textPt, xf)

        Case "AcDbMLeader"
            ' Limited MLeader API: text is reliably accessible on text-content leaders.
            ' Leader-line vertices and arrowheads vary by AutoCAD version and style;
            ' only output an MLeader after extracting geometry; otherwise warn.
            TSA_ExportMLeader ent, row, out, xf, exportLayer
            Exit Sub

        Case "AcDbHatch"
            TSA_CaptureHatch ent, exportLayer, xf
            Exit Sub

        Case "AcDbArc"
            row.Add "type", "ARC"
            row.Add "centre", TSA_Point(ent.Center, xf)
            row.Add "radius", CDbl(ent.Radius) * TSA_ApproxScale(xf)
            row.Add "startAngle", CDbl(ent.StartAngle)
            row.Add "endAngle", CDbl(ent.EndAngle)
            TSA_Warn exportLayer, ent.Handle, typ, "ARC angles are in native entity plane, may need transformed approximation."

        Case Else
            TSA_Warn exportLayer, ent.Handle, typ, "Unsupported type; excluded from JSON."
            Exit Sub
    End Select
    If Not TSA_ValidEntityCoordinates(row, exportLayer, ent.Handle) Then
        'OCS and 3D polyline fallback points are already in section-space UCS.
        If (typ = "AcDbPolyline" Or typ = "AcDb2dPolyline" Or typ = "AcDb3dPolyline") Then
            Dim fallbackPts As Collection
            Set fallbackPts = TSA_FallbackPoints(ent.Handle)
            If Not fallbackPts Is Nothing Then
                Set row("points") = fallbackPts
                If TSA_ValidEntityCoordinates(row, exportLayer, ent.Handle) Then
                    out.Add row
                    Exit Sub
                End If
            End If
        End If
        Exit Sub
    End If
    out.Add row
End Sub

Private Sub TSA_ExportMLeader(ByVal ent As AcadEntity, ByVal row As Object, _
                              ByRef out As Collection, ByVal xf As Variant, ByVal layerName As String)
    Dim ml As AcadMLeader, leaderNumber As Long, leaderLineID As Variant
    Dim lineIDs As Variant, coords As Variant, pts As Collection
    Dim j As Long, hasGeometry As Boolean, textValue As String
    Dim leader As Object, lo As Variant, hi As Variant, approxTextPoint As Variant
    Dim textBoxOk As Boolean, clusterCount As Long
    Dim lastLeaderPoint As Collection, approximatePos As Collection

    'Stage 1.7: DXF-context fallback replaces BOTH provisional ActiveX parts.
    'Use only when both a trusted text point and a leader line exist;
    'otherwise retain Stage 1.5 behaviour for any partial result.
    Dim dxfTextPoint As Collection, dxfLeaderPoints As Collection
    Dim dxfLine As Object
    Set dxfTextPoint = TSA_FallbackTextPoint(ent.Handle)
    Set dxfLeaderPoints = TSA_FallbackLeaderPoints(ent.Handle)
    If Not dxfTextPoint Is Nothing And Not dxfLeaderPoints Is Nothing Then
        If dxfLeaderPoints.Count >= 2 Then
            Set ml = ent
            Set dxfLine = TSA_BaseEntity(ent, layerName)
            dxfLine.Add "type", "POLYLINE"
            dxfLine.Add "points", dxfLeaderPoints
            dxfLine.Add "arrowAtStart", True
            dxfLine.Add "mleaderPart", "leaderLine"
            If TSA_ValidEntityCoordinates(dxfLine, layerName, ent.Handle) Then
                out.Add dxfLine
                textValue = ml.TextString
                If Len(textValue) > 0 Then
                    row.Add "type", "MTEXT"
                    row.Add "text", textValue
                    row.Add "position", dxfTextPoint
                    row.Add "height", CDbl(ml.TextHeight) * TSA_ApproxScale(xf)
                    row.Add "positionApproximate", False
                    row.Add "mleaderPart", "text"
                    If TSA_ValidEntityCoordinates(row, layerName, ent.Handle) Then out.Add row
                End If
                Exit Sub
            End If
        End If
    End If

    Set ml = ent
    'ActiveX exposes LeaderCount, not LeaderLineCount. Each leader cluster
    'has its own set of line indexes (which need not be 0,1,2,...).
    On Error GoTo LeaderError
    clusterCount = ml.LeaderCount
    For leaderNumber = 0 To clusterCount - 1
        'LeaderCount is a count, not an index inventory. Some drawings use
        'non-contiguous leader IDs; invalid entries must not prevent text export.
        On Error Resume Next
        Err.Clear
        lineIDs = ml.GetLeaderLineIndexes(leaderNumber)
        If Err.Number <> 0 Then
            TSA_Warn layerName, ent.Handle, ent.ObjectName, _
                "MLeader cluster index " & CStr(leaderNumber) & _
                " inaccessible: " & Err.Description & "; leader geometry omitted."
            Err.Clear
            On Error GoTo LeaderError
            GoTo NextLeaderCluster
        End If
        On Error GoTo LeaderError
        If IsArray(lineIDs) Then
            For Each leaderLineID In lineIDs
                On Error Resume Next
                Err.Clear
                coords = ml.GetLeaderLineVertices(CLng(leaderLineID))
                If Err.Number <> 0 Then
                    TSA_Warn layerName, ent.Handle, ent.ObjectName, _
                             "MLeader line index " & CStr(leaderLineID) & _
                             " inaccessible: " & Err.Description
                    Err.Clear
                    On Error GoTo LeaderError
                    GoTo NextLeaderLine
                End If
                On Error GoTo LeaderError
                If IsArray(coords) Then
                    Set pts = New Collection
                    For j = LBound(coords) To UBound(coords) - 2 Step 3
                        pts.Add TSA_Point(TSA_XYZ(CDbl(coords(j)), CDbl(coords(j + 1)), CDbl(coords(j + 2))), xf)
                    Next j
                    If pts.Count >= 2 Then
                        Set leader = TSA_BaseEntity(ent, layerName)
                        leader.Add "type", "POLYLINE"
                        leader.Add "points", pts
                        leader.Add "arrowAtStart", True
                        leader.Add "mleaderPart", "leaderLine"
                        If TSA_ValidEntityCoordinates(leader, layerName, ent.Handle) Then
                            out.Add leader
                            Set lastLeaderPoint = pts(pts.Count)
                            hasGeometry = True
                        End If
                    End If
                End If
NextLeaderLine:
            Next leaderLineID
        End If
NextLeaderCluster:
    Next leaderNumber
AfterLeaderLines:
    On Error GoTo TextError
    textValue = ml.TextString
    If Len(textValue) > 0 Then
        'Do not use the entity bounding-box centre: in this drawing it can
        'be WCS Easting/Northing and is not a reliable section text anchor.
        'Prefer a section-space leader endpoint when one was extracted.
        If Not lastLeaderPoint Is Nothing Then
            Set approximatePos = New Collection
            approximatePos.Add CDbl(lastLeaderPoint(1))
            approximatePos.Add CDbl(lastLeaderPoint(2))
            row.Add "type", "MTEXT"
            row.Add "text", textValue
            Dim exactTextPt As Collection
            Set exactTextPt = TSA_FallbackTextPoint(ent.Handle)
            If Not exactTextPt Is Nothing Then Set approximatePos = exactTextPt
            row.Add "position", approximatePos
            row.Add "height", CDbl(ml.TextHeight) * TSA_ApproxScale(xf)
            row.Add "positionApproximate", (exactTextPt Is Nothing)
            row.Add "mleaderPart", "text"
            out.Add row
            hasGeometry = True
            If exactTextPt Is Nothing Then
                TSA_Warn layerName, ent.Handle, ent.ObjectName, _
                    "MLeader text anchored near leader endpoint; position is provisional."
            End If
        Else
            Set approximatePos = TSA_FallbackTextPoint(ent.Handle)
            If Not approximatePos Is Nothing Then
                row.Add "type", "MTEXT"
                row.Add "text", textValue
                row.Add "position", approximatePos
                row.Add "height", CDbl(ml.TextHeight) * TSA_ApproxScale(xf)
                row.Add "positionApproximate", False
                row.Add "mleaderPart", "text"
                If TSA_ValidEntityCoordinates(row, layerName, ent.Handle) Then
                    out.Add row
                    hasGeometry = True
                End If
            Else
                TSA_Warn layerName, ent.Handle, ent.ObjectName, _
                    "MLeader text preserved but omitted: no section-space anchor. Text: " & Left$(textValue, 80)
            End If
        End If
    End If
AfterText:
    On Error GoTo 0
    If Not hasGeometry Then
        TSA_Warn layerName, ent.Handle, ent.ObjectName, _
                 "MLeader has no successfully exported vertices/text; inspect source entity."
    Else
        TSA_Warn layerName, ent.Handle, ent.ObjectName, _
                 "MLeader export is provisional: arrowhead, dogleg and spline fidelity are not exact."
    End If
    Exit Sub
LeaderError:
    TSA_Warn layerName, ent.Handle, ent.ObjectName, _
             "MLeader leader-line read failed: " & Err.Description
    Err.Clear
    Resume AfterLeaderLines
TextError:
    TSA_Warn layerName, ent.Handle, ent.ObjectName, _
             "MLeader text extraction failed: " & Err.Description
    Err.Clear
    Resume AfterText
End Sub

Private Function TSA_HatchCircleBoundaryHandles(ByVal hatchEntity As AcadEntity) As Object
    ' Associative AutoCAD hatches retain references to their boundary objects.
    ' Match these by entity HANDLE rather than guessing from a WCS bounding box.
    ' Non-associative hatches may not expose any boundary objects; return empty.
    Dim found As Object, hatchObj As Object, loopItems As Variant
    Dim item As Variant, k As Long, loopCount As Long, h As String
    Set found = CreateObject("Scripting.Dictionary")
    found.CompareMode = vbTextCompare
    On Error GoTo Done
    Set hatchObj = hatchEntity
    loopCount = CLng(hatchObj.NumberOfLoops)
    For k = 0 To loopCount - 1
        loopItems = Empty
        On Error Resume Next
        hatchObj.GetLoopAt k, loopItems
        If Err.Number <> 0 Then
            Err.Clear
            On Error GoTo Done
            GoTo NextLoop
        End If
        On Error GoTo Done
        If IsArray(loopItems) Then
            For Each item In loopItems
                On Error Resume Next
                If StrComp(CStr(item.ObjectName), "AcDbCircle", vbTextCompare) = 0 Then
                    h = CStr(item.Handle)
                    If Not found.Exists(h) Then found.Add h, True
                End If
                Err.Clear
                On Error GoTo Done
            Next item
        End If
NextLoop:
    Next k
Done:
    Set TSA_HatchCircleBoundaryHandles = found
End Function

Private Sub TSA_CaptureHatch(ByVal ent As AcadEntity, ByVal layerName As String, ByVal xf As Variant)
    Dim hatch As Object, lo As Variant, hi As Variant
    Dim pat As String
    pat = CStr(ent.PatternName)
    If StrComp(pat, "SOLID", vbTextCompare) <> 0 Then
        TSA_Warn layerName, ent.Handle, ent.ObjectName, "Non-solid hatch pattern '" & pat & "' not exported."
        Exit Sub
    End If
    On Error GoTo CannotRead
    ent.GetBoundingBox lo, hi
    Set hatch = TSA_BaseEntity(ent, layerName)
    hatch.Add "min", TSA_Point(lo, xf)
    hatch.Add "max", TSA_Point(hi, xf)
    hatch.Add "consumed", False
    hatch.Add "circleBoundaryHandles", TSA_HatchCircleBoundaryHandles(ent)
    mHatches.Add hatch
    Exit Sub
CannotRead:
    TSA_Warn layerName, ent.Handle, ent.ObjectName, "Cannot extract hatch extents: " & Err.Description
End Sub

Private Sub TSA_MatchHatchCircles(ByRef out As Collection, ByVal layerName As String)
    Dim c As Variant, h As Variant, boundary As Object
    Dim ctr As Collection, lower As Collection, upper As Collection
    Dim radius As Double, diameter As Double, dx As Double, dy As Double
    Dim cX As Double, cY As Double, matched As Boolean
    Dim cHandle As String

    For Each c In mCircles
        matched = False
        cHandle = CStr(c("handle"))
        If TSA_IsBoreholeLayer(layerName) Then c("colour") = "#000000"

        'First try an exact match using AutoCAD's associative hatch boundary.
        'This avoids the false mismatches caused by rotated/3D WCS bounding boxes.
        For Each h In mHatches
            If h("consumed") = False Then
                Set boundary = h("circleBoundaryHandles")
                If boundary.Exists(cHandle) Then
                    c("fill") = CStr(h("colour"))
                    h("consumed") = True
                    matched = True
                    Exit For
                End If
            End If
        Next h

        'Fallback for non-associative hatches: original strict dimension/centre match.
        'Do not assign a fill just because a hatch happens to be nearby.
        If Not matched Then
            If Not c.Exists("centre") Or Not c.Exists("radius") Then
                TSA_Warn layerName, cHandle, "CIRCLE", "Circle missing centre/radius; hatch match skipped."
            Else
                Set ctr = c("centre")
                If ctr.Count >= 2 Then
                    radius = CDbl(c("radius"))
                    diameter = radius * 2#
                    For Each h In mHatches
                        If h("consumed") = False Then
                            Set lower = h("min")
                            Set upper = h("max")
                            If lower.Count >= 2 And upper.Count >= 2 Then
                                dx = Abs(CDbl(upper(1)) - CDbl(lower(1)))
                                dy = Abs(CDbl(upper(2)) - CDbl(lower(2)))
                                cX = (CDbl(lower(1)) + CDbl(upper(1))) / 2#
                                cY = (CDbl(lower(2)) + CDbl(upper(2))) / 2#
                                If Abs(dx - diameter) < TSA_CIRCLE_MATCH_TOLERANCE And _
                                   Abs(dy - diameter) < TSA_CIRCLE_MATCH_TOLERANCE And _
                                   Abs(cX - CDbl(ctr(1))) < TSA_CIRCLE_MATCH_TOLERANCE And _
                                   Abs(cY - CDbl(ctr(2))) < TSA_CIRCLE_MATCH_TOLERANCE Then
                                    c("fill") = CStr(h("colour"))
                                    h("consumed") = True
                                    matched = True
                                    Exit For
                                End If
                            End If
                        End If
                    Next h
                End If
            End If
        End If
        If TSA_IsBoreholeLayer(layerName) And Not matched Then
            TSA_Warn layerName, cHandle, "AcDbCircle", _
                "No matching solid hatch: circle exported black-outlined with no fill."
        End If
    Next c
    For Each h In mHatches
        If h("consumed") = False Then
            TSA_Warn layerName, CStr(h("handle")), "AcDbHatch", _
                     "Unmatched solid hatch; cannot safely reconstruct its boundary."
        End If
    Next h
End Sub


' ===== XCPosition cross-cut blocks =====
' The CAD diagnostic confirms that XCPosition contains a section-plane
' 41.8651-unit centreline (local X=0, Y=0..41.8651) and two live attributes:
' ROADNAME and CHAINAGE. Generic block expansion rotates its section-plane
' vertical into a horizontal WCS curve. Work in CHAINAGE_ELEVATION here.
Private Sub TSA_ExportXCPosition(ByVal br As AcadBlockReference, _
    ByVal blockDef As AcadBlock, ByVal childXf As Variant, _
    ByRef out As Collection, ByVal exportLayer As String)
    Dim child As AcadEntity, coords As Variant, p0 As Collection
    Dim centreX As Double, centreY As Double, localY As Double
    Dim lineLength As Double, sy As Double, row As Object, geom As Collection
    Dim attrs As Variant, att As Variant, tag As String, value As String
    Dim nameValue As String, chainValue As String, centreHandle As String
    Dim baseY As Double, y0 As Double, y1 As Double

    lineLength = 0#
    centreHandle = br.Handle
    For Each child In blockDef
        If child.ObjectName = "AcDbPolyline" Then
            coords = child.Coordinates
            If UBound(coords) >= 3 Then
                lineLength = Abs(CDbl(coords(3)) - CDbl(coords(1)))
                localY = CDbl(coords(1))
                Set p0 = TSA_Point(TSA_XYZ(CDbl(coords(0)), localY, _
                    TSA_EntityElevation(child)), childXf)
                centreHandle = child.Handle
                Exit For
            End If
        End If
    Next child
    If lineLength <= 0# Then
        TSA_Warn exportLayer, br.Handle, "XCPosition", "No centreline geometry in XCPosition definition."
        Exit Sub
    End If
    centreX = CDbl(p0(1))
    centreY = CDbl(p0(2))
    sy = br.YScaleFactor
    If sy = 0# Then sy = 1#
    y0 = centreY: y1 = centreY + lineLength * sy

    Set row = TSA_Dict()
    row.Add "handle", CStr(centreHandle)
    row.Add "parentBlockHandle", CStr(br.Handle)
    row.Add "blockName", "XCPosition"
    row.Add "layer", exportLayer
    row.Add "originalLayer", CStr(child.Layer)
    row.Add "colour", "#0000FF"
    row.Add "linetype", "ByLayer"
    row.Add "lineweight", -1
    row.Add "type", "POLYLINE"
    Set geom = New Collection
    geom.Add TSA_XCPoint(centreX, y0)
    geom.Add TSA_XCPoint(centreX, y1)
    row.Add "points", geom
    row.Add "closed", False
    out.Add row

    nameValue = "": chainValue = ""
    If br.HasAttributes Then
        attrs = br.GetAttributes
        For Each att In attrs
            tag = UCase$(Trim$(CStr(att.TagString)))
            value = CStr(att.TextString)
            If tag = "ROADNAME" Then nameValue = value
            If tag = "CHAINAGE" Then chainValue = value
        Next att
    End If
    'Position labels near the upper end, matching their definition's offsets:
    'ROADNAME (0.199428,39.8997); CHAINAGE (0.140983,37.1914).
    If Len(nameValue) > 0 Then _
        TSA_AppendXCText out, exportLayer, br.Handle, "ROADNAME", nameValue, _
            centreX + 0.199428, centreY + 39.8997 * sy
    If Len(chainValue) > 0 Then _
        TSA_AppendXCText out, exportLayer, br.Handle, "CHAINAGE", chainValue, _
            centreX + 0.140983, centreY + 37.1914 * sy
End Sub

' ===== Geometry and block transforms =====
' 2D affine transform over CAD XY, retaining Z separately:
' [a,b,c,d,tx,ty,sz,tz] where x'=a*x+c*y+tx, y'=b*x+d*y+ty, z'=sz*z+tz
Private Function TSA_Identity() As Variant
    TSA_Identity = Array(1#, 0#, 0#, 1#, 0#, 0#, 1#, 0#)
End Function

Private Function TSA_BlockTransform(ByVal br As AcadBlockReference) As Variant
    Dim a As Double, sx As Double, sy As Double, ins As Variant
    a = br.Rotation: sx = br.XScaleFactor: sy = br.YScaleFactor
    ins = br.InsertionPoint
    TSA_BlockTransform = Array(sx * Cos(a), sx * Sin(a), -sy * Sin(a), sy * Cos(a), _
                               CDbl(ins(0)), CDbl(ins(1)), CDbl(br.ZScaleFactor), CDbl(ins(2)))
End Function

Private Function TSA_Translation(ByVal x As Double, ByVal y As Double, ByVal z As Double) As Variant
    TSA_Translation = Array(1#, 0#, 0#, 1#, x, y, 1#, z)
End Function

Private Function TSA_Compose(ByVal parent As Variant, ByVal child As Variant) As Variant
    TSA_Compose = Array( _
        parent(0) * child(0) + parent(2) * child(1), _
        parent(1) * child(0) + parent(3) * child(1), _
        parent(0) * child(2) + parent(2) * child(3), _
        parent(1) * child(2) + parent(3) * child(3), _
        parent(0) * child(4) + parent(2) * child(5) + parent(4), _
        parent(1) * child(4) + parent(3) * child(5) + parent(5), _
        parent(6) * child(6), _
        parent(6) * child(7) + parent(7))
End Function

Private Function TSA_XYZ(ByVal x As Double, ByVal y As Double, ByVal z As Double) As Variant
    TSA_XYZ = Array(x, y, z)
End Function

Private Function TSA_TransformWCS(ByVal xyz As Variant, ByVal xf As Variant) As Variant
    TSA_TransformWCS = Array( _
        xf(0) * CDbl(xyz(0)) + xf(2) * CDbl(xyz(1)) + xf(4), _
        xf(1) * CDbl(xyz(0)) + xf(3) * CDbl(xyz(1)) + xf(5), _
        xf(6) * CDbl(xyz(2)) + xf(7))
End Function

Private Function TSA_Point(ByVal xyz As Variant, ByVal xf As Variant) As Collection
    Dim w As Variant, u As Variant, pt As Collection
    Dim cadPoint(0 To 2) As Double
    w = TSA_TransformWCS(xyz, xf)
    'AutoCAD requires SAFEARRAY(Double), not the SAFEARRAY(Variant) produced by Array().
    cadPoint(0) = CDbl(w(0))
    cadPoint(1) = CDbl(w(1))
    cadPoint(2) = CDbl(w(2))
    u = mDoc.Utility.TranslateCoordinates(cadPoint, acWorld, acUCS, False)
    Set pt = New Collection
    pt.Add CDbl(u(0))
    pt.Add CDbl(u(1))
    Set TSA_Point = pt
End Function

Private Function TSA_AngleInSection(ByVal origin As Variant, ByVal angleRad As Double, ByVal xf As Variant) As Double
    Dim endpoint As Variant, p As Collection, q As Collection
    endpoint = TSA_XYZ(CDbl(origin(0)) + Cos(angleRad), CDbl(origin(1)) + Sin(angleRad), CDbl(origin(2)))
    Set p = TSA_Point(origin, xf): Set q = TSA_Point(endpoint, xf)
    TSA_AngleInSection = Atn2(CDbl(q(2)) - CDbl(p(2)), CDbl(q(1)) - CDbl(p(1)))
End Function

Private Function Atn2(ByVal y As Double, ByVal x As Double) As Double
    If x = 0# Then
        If y >= 0# Then Atn2 = 2# * Atn(1#) Else Atn2 = -2# * Atn(1#)
    Else
        Atn2 = Atn(y / x)
        If x < 0 Then Atn2 = Atn2 + IIf(y >= 0#, 4# * Atn(1#), -4# * Atn(1#))
    End If
End Function

Private Function TSA_ApproxScale(ByVal xf As Variant) As Double
    Dim sx As Double, sy As Double
    sx = Sqr(xf(0) ^ 2 + xf(1) ^ 2)
    sy = Sqr(xf(2) ^ 2 + xf(3) ^ 2)
    TSA_ApproxScale = (sx + sy) / 2#
End Function

'Conservative QA: a stray coordinate must not distort the section viewport.
'Values are generous and based on a practical 0-20 km chainage / 0-2000 m RL.
Private Function TSA_ValidEntityCoordinates(ByVal row As Object, _
    ByVal layerName As String, ByVal handle As String) As Boolean
    Dim pts As Collection, pt As Collection, k As Variant
    Dim i As Long, typ As String
    TSA_ValidEntityCoordinates = True
    typ = CStr(row("type"))
    Select Case typ
        Case "LINE", "POLYLINE"
            Set pts = row("points")
            For i = 1 To pts.Count
                Set pt = pts(i)
                If Not TSA_ValidXY(CDbl(pt(1)), CDbl(pt(2))) Then GoTo InvalidPoint
            Next i
        Case "TEXT", "MTEXT"
            Set pt = row("position")
            If Not TSA_ValidXY(CDbl(pt(1)), CDbl(pt(2))) Then GoTo InvalidPoint
        Case "CIRCLE", "ARC"
            Set pt = row("centre")
            If Not TSA_ValidXY(CDbl(pt(1)), CDbl(pt(2))) Then GoTo InvalidPoint
    End Select
    Exit Function
InvalidPoint:
    TSA_ValidEntityCoordinates = False
    TSA_Warn layerName, handle, typ, _
        "Geometry outside section-space bounds in VBA; checking fallback if supplied."
End Function

Private Function TSA_ValidXY(ByVal x As Double, ByVal y As Double) As Boolean
    TSA_ValidXY = (x > -10000# And x < 20000# And y > -2000# And y < 3000#)
End Function

'Approximate each bulged LWPolyline segment by short chords in source plane.
'AutoCAD bulge = tan(includedAngle / 4). Replaces only when actual bulges exist.
Private Sub TSA_AppendBulgePoints(ByVal ent As AcadEntity, ByRef pts As Collection, _
    ByVal xf As Variant, ByVal layerName As String)
    Dim coords As Variant, rawPts As Collection, result As Collection
    Dim i As Long, j As Long, n As Long, segmentCount As Long, nSteps As Long
    Dim bulge As Double, x1 As Double, y1 As Double, x2 As Double, y2 As Double
    Dim mx As Double, my As Double, dx As Double, dy As Double, length As Double
    Dim theta As Double, radius As Double, offset As Double, cx As Double, cy As Double
    Dim angle0 As Double, a As Double, t As Double, z As Double
    Dim p As Variant, isClosed As Boolean, changed As Boolean
    On Error GoTo BulgeFail
    coords = ent.Coordinates
    n = (UBound(coords) - LBound(coords) + 1) \ 2
    If n < 2 Then Exit Sub
    isClosed = TSA_EntityClosed(ent)
    segmentCount = n - 1
    If isClosed Then segmentCount = n
    Set result = New Collection
    z = TSA_EntityElevation(ent)
    For i = 0 To segmentCount - 1
        j = (i + 1) Mod n
        x1 = CDbl(coords(2 * i)): y1 = CDbl(coords(2 * i + 1))
        x2 = CDbl(coords(2 * j)): y2 = CDbl(coords(2 * j + 1))
        result.Add TSA_Point(TSA_XYZ(x1, y1, z), xf)
        bulge = CDbl(ent.GetBulge(i))
        If Abs(bulge) > 0.0000001 Then
            changed = True
            dx = x2 - x1: dy = y2 - y1
            length = Sqr(dx * dx + dy * dy)
            If length > 0.000001 Then
                theta = 4# * Atn(bulge)
                radius = length * (1# + bulge * bulge) / (4# * Abs(bulge))
                offset = length * (1# - bulge * bulge) / (4# * bulge)
                mx = (x1 + x2) / 2#: my = (y1 + y2) / 2#
                cx = mx - dy / length * offset
                cy = my + dx / length * offset
                angle0 = Atn2(y1 - cy, x1 - cx)
                nSteps = CLng(Abs(theta) * radius / 0.25) + 1
                If nSteps < 4 Then nSteps = 4
                If nSteps > 256 Then nSteps = 256
                For j = 1 To nSteps - 1
                    a = angle0 + theta * CDbl(j) / CDbl(nSteps)
                    result.Add TSA_Point(TSA_XYZ(cx + radius * Cos(a), cy + radius * Sin(a), z), xf)
                Next j
            End If
        End If
    Next i
    If Not isClosed Then
        result.Add TSA_Point(TSA_XYZ(CDbl(coords(2 * (n - 1))), _
            CDbl(coords(2 * (n - 1) + 1)), z), xf)
    End If
    If changed Then Set pts = result
    Exit Sub
BulgeFail:
    TSA_Warn layerName, ent.Handle, ent.ObjectName, _
        "Polyline arc interpolation failed; retained straight chords: " & Err.Description
End Sub


' ===== Optional AutoLISP fallback bridge =====
' Run TSAOVLFALLBACK in AutoCAD first; pass its file to the new entrypoint.
Private Sub TSA_LoadFallback(ByVal path As String)
    Dim ff As Integer, line As String, parts As Variant
    Set mFallback = CreateObject("Scripting.Dictionary")
    mFallback.CompareMode = vbTextCompare
    If Len(Trim$(path)) = 0 Then Exit Sub
    If Len(Dir$(path)) = 0 Then
        Err.Raise vbObjectError + 7080, , "Fallback file not found: " & path
    End If
    ff = FreeFile
    Open path For Input As #ff
    Do While Not EOF(ff)
        Line Input #ff, line
        parts = Split(line, "|", 3)
        If UBound(parts) = 2 Then
            mFallback(CStr(parts(0)) & "|" & CStr(parts(1))) = CStr(parts(2))
        End If
    Loop
    Close #ff
End Sub

Private Function TSA_ParseSectionPoints(ByVal src As String) As Collection
    Dim pairs As Variant, pair As Variant, xy As Variant, p As Collection
    Dim result As Collection
    Set result = New Collection
    pairs = Split(src, ";")
    For Each pair In pairs
        xy = Split(CStr(pair), ",")
        If UBound(xy) <> 1 Then Exit Function
        Set p = New Collection
        p.Add Val(CStr(xy(0)))
        p.Add Val(CStr(xy(1)))
        result.Add p
    Next pair
    Set TSA_ParseSectionPoints = result
End Function

Private Function TSA_FallbackPoints(ByVal handle As String) As Collection
    If mFallback Is Nothing Then Exit Function
    If mFallback.Exists("P|" & handle) Then
        Set TSA_FallbackPoints = TSA_ParseSectionPoints(mFallback("P|" & handle))
    End If
End Function

Private Function TSA_FallbackLeaderPoints(ByVal handle As String) As Collection
    If mFallback Is Nothing Then Exit Function
    If mFallback.Exists("L|" & handle) Then
        Set TSA_FallbackLeaderPoints = TSA_ParseSectionPoints(mFallback("L|" & handle))
    End If
End Function

Private Function TSA_FallbackTextPoint(ByVal handle As String) As Collection
    Dim pts As Collection
    If mFallback Is Nothing Then Exit Function
    If mFallback.Exists("M|" & handle) Then
        Set pts = TSA_ParseSectionPoints(mFallback("M|" & handle))
        If Not pts Is Nothing Then
            If pts.Count = 1 Then Set TSA_FallbackTextPoint = pts(1)
        End If
    End If
End Function

' Stage 2 - Upload a complete Stage 1.7 CAD overlay JSON snapshot to Supabase.
' Depends on JsonConverter.bas and TSA_CADExportLayers() from Stage 1.7.
' The expectedRdID is mandatory and protects against an incorrect export argument.
' Upserts one row per (RdID, layerName); empty arrays clear an existing layer.
' No deletes, no edits to GEO_SectionFeatures.
'
' Example after EnsureSupabaseToken:
' TSA_UploadCADOverlayJSON "C:\Temp\TSA_section_overlay.json", _
'     "https://YOUR-PROJECT.supabase.co", "YOUR-PUBLISHABLE-KEY", _
'     SupabaseAccessToken, 8729
'
Public Sub TSA_UploadCADOverlayJSON( _
    ByVal filePath As String, ByVal supabaseUrl As String, _
    ByVal publishableKey As String, ByVal accessToken As String, _
    ByVal expectedRdID As Long)

    On Error GoTo UploadFailure
    Dim stm As Object, root As Object, section As Object
    Dim layers As Collection, layer As Variant, entities As Collection
    Dim rows As Collection, seen As Object, rawLayers As Collection, rawLayer As String
    Dim cfg As Variant, conf As Variant, name As String
    Dim rd As Long, drawing As String, rev As String
    Dim warnCount As Long, raw As String, endpoint As String
    Dim request As Object, result As String

    If expectedRdID <= 0 Then Err.Raise vbObjectError + 7400, , "Expected RdID must be positive."
    If Len(Trim$(publishableKey)) = 0 Or Len(Trim$(accessToken)) = 0 Then _
        Err.Raise vbObjectError + 7401, , "Logged-in user's access token and publishable key are required."
    If Len(Trim$(supabaseUrl)) = 0 Then Err.Raise vbObjectError + 7402, , "Supabase project URL required."

    Set stm = CreateObject("ADODB.Stream")
    stm.Type = 2
    stm.Charset = "utf-8"
    stm.Open
    stm.LoadFromFile filePath
    raw = stm.ReadText(-1)
    stm.Close
    Set stm = Nothing
    If Len(Trim$(raw)) = 0 Then Err.Raise vbObjectError + 7403, , "Overlay JSON file is empty."
    ' ADODB.Stream normally consumes UTF-8 BOM, but guard against retained BOM.
    If Left$(raw, 1) = ChrW$(&HFEFF) Then raw = Mid$(raw, 2)

    Set root = JsonConverter.ParseJson(raw)
    If CStr(root("source")) <> "CAD_OVERLAY" Then Err.Raise vbObjectError + 7404, , "Unexpected JSON source."
    If CLng(root("formatVersion")) <> 1 Then Err.Raise vbObjectError + 7405, , "Unsupported formatVersion."
    Set section = root("section")
    rd = CLng(section("RdID"))
    If rd <> expectedRdID Then Err.Raise vbObjectError + 7406, , _
        "Roadway ID mismatch. File RdID=" & rd & "; expected=" & expectedRdID & ". Regenerate export with the correct RdID."
    If CStr(section("coordinateSystem")) <> "CHAINAGE_ELEVATION" Then _
        Err.Raise vbObjectError + 7407, , "Expected CHAINAGE_ELEVATION coordinates."
    If CStr(section("units")) <> "metres" Then Err.Raise vbObjectError + 7408, , "Expected metres."
    drawing = Trim$(CStr(root("sourceDrawing")))
    rev = Trim$(CStr(root("sourceRevision")))
    If Len(drawing) = 0 Or Len(rev) = 0 Then Err.Raise vbObjectError + 7409, , "Drawing/revision missing."

    Set layers = root("layers")
    Set seen = CreateObject("Scripting.Dictionary")
    seen.CompareMode = vbTextCompare
    Set rows = New Collection
    Set rawLayers = TSA_RawEntityArrays(raw)
    If rawLayers.Count <> layers.Count Then Err.Raise vbObjectError + 7420, , "Raw JSON layer/array count mismatch."
    conf = TSA_CADExportLayers()

    Dim layerIndex As Long
    layerIndex = 0
    For Each layer In layers
        layerIndex = layerIndex + 1
        name = CStr(layer("layerName"))
        If seen.Exists(name) Then Err.Raise vbObjectError + 7410, , "Duplicate layer in export: " & name
        Dim configured As Boolean
        configured = False
        For Each cfg In conf
            If StrComp(name, CStr(cfg), vbTextCompare) = 0 Then configured = True
        Next cfg
        If Not configured Then Err.Raise vbObjectError + 7411, , "Unexpected layer: " & name
        If Not CBool(layer("layerExists")) Then _
            Err.Raise vbObjectError + 7412, , "Configured drawing layer missing: " & name
        Set entities = layer("entities")
        rawLayer = CStr(rawLayers(layerIndex))
        rows.Add "{" & """RdID"":" & CStr(rd) & "," & _
            """layerName"":" & JsonConverter.ConvertToJson(name) & "," & _
            """entities"":" & rawLayer & "," & _
            """sourceDrawing"":" & JsonConverter.ConvertToJson(drawing) & "," & _
            """sourceRevision"":" & JsonConverter.ConvertToJson(rev) & "}"
        seen.Add name, True
    Next layer
    For Each cfg In conf
        If Not seen.Exists(CStr(cfg)) Then Err.Raise vbObjectError + 7413, , "Missing layer: " & CStr(cfg)
    Next cfg
    If rows.Count = 0 Then Err.Raise vbObjectError + 7414, , "No layers to upload."

    warnCount = root("warnings").Count
    Debug.Print "Overlay upload RdID=" & rd & ", layers=" & rows.Count & _
                ", extraction notices=" & warnCount
    If warnCount > 0 Then Debug.Print "Review local JSON warnings before using overlay operationally."

    If Right$(supabaseUrl, 1) = "/" Then supabaseUrl = Left$(supabaseUrl, Len(supabaseUrl) - 1)
    endpoint = supabaseUrl & "/rest/v1/GEO_SectionCADOverlays?on_conflict=RdID%2ClayerName"
    Set request = CreateObject("MSXML2.ServerXMLHTTP.6.0")
    request.setTimeouts 15000, 30000, 120000, 240000
    request.Open "POST", endpoint, False
    request.setRequestHeader "apikey", publishableKey
    request.setRequestHeader "Authorization", "Bearer " & accessToken
    request.setRequestHeader "Content-Type", "application/json; charset=utf-8"
    request.setRequestHeader "Prefer", "resolution=merge-duplicates,return=minimal"
    request.send TSA_RawRowsToJSON(rows)
    result = request.responseText
    Debug.Print "Overlay upload HTTP status: " & request.Status
    If request.Status < 200 Or request.Status >= 300 Then _
        Err.Raise vbObjectError + 7415, , "Supabase HTTP " & request.Status & ": " & Left$(result, 1800)

    MsgBox "CAD overlay snapshots uploaded successfully." & vbCrLf & _
           "Roadway: " & rd & vbCrLf & "Layers: " & rows.Count & vbCrLf & _
           "Local extraction notices: " & warnCount, vbInformation
    Exit Sub

UploadFailure:
    Dim errorText As String
    errorText = Err.Description
    On Error Resume Next
    If Not stm Is Nothing Then stm.Close
    On Error GoTo 0
    Err.Raise vbObjectError + 7421, "TSA_UploadCADOverlayJSON", "CAD overlay upload failed: " & errorText
End Sub

' =============================================================
' RAW JSON TRANSFER: DO NOT re-serialize coordinates with VBA-JSON.
' VBA-JSON can stringify high-precision negative decimal values.
' Validate parsed JSON above, but pass entities verbatim to PostgREST.
' =============================================================
Private Function TSA_RawEntityArrays(ByVal json As String) As Collection
    Dim results As New Collection, layersAt As Long, p As Long, q As Long
    layersAt = InStr(1, json, """layers""", vbBinaryCompare)
    If layersAt = 0 Then Err.Raise vbObjectError + 7430, , "Missing layers key."
    p = InStr(layersAt, json, "[")
    If p = 0 Then Err.Raise vbObjectError + 7431, , "Missing layers array."
    Do
        q = TSA_JSONSkipSpace(json, p + 1)
        If q > Len(json) Then Err.Raise vbObjectError + 7432, , "Unterminated layers array."
        If Mid$(json, q, 1) = "]" Then Exit Do
        If Mid$(json, q, 1) = "," Then
            p = q
        ElseIf Mid$(json, q, 1) = "{" Then
            Dim endObject As Long, obj As String, entityKey As Long, arrayPos As Long
            endObject = TSA_JSONMatchingEnd(json, q)
            obj = Mid$(json, q, endObject - q + 1)
            entityKey = InStr(1, obj, """entities""", vbBinaryCompare)
            If entityKey = 0 Then Err.Raise vbObjectError + 7433, , "Layer lacks entities."
            arrayPos = InStr(entityKey + 10, obj, "[")
            If arrayPos = 0 Then Err.Raise vbObjectError + 7434, , "Layer entities is not array."
            Dim arrayEnd As Long
            arrayEnd = TSA_JSONMatchingEnd(obj, arrayPos)
            results.Add Mid$(obj, arrayPos, arrayEnd - arrayPos + 1)
            p = endObject + 1
        Else
            Err.Raise vbObjectError + 7435, , "Unexpected layers content."
        End If
    Loop
    Set TSA_RawEntityArrays = results
End Function

Private Function TSA_JSONSkipSpace(ByVal json As String, ByVal startPos As Long) As Long
    Dim c As String
    Do While startPos <= Len(json)
        c = Mid$(json, startPos, 1)
        If c <> " " And c <> vbTab And c <> vbCr And c <> vbLf Then Exit Do
        startPos = startPos + 1
    Loop
    TSA_JSONSkipSpace = startPos
End Function

Private Function TSA_JSONMatchingEnd(ByVal src As String, ByVal startPos As Long) As Long
    Dim i As Long, depth As Long, c As String, inString As Boolean, escaped As Boolean
    Dim opening As String, closing As String
    opening = Mid$(src, startPos, 1)
    If opening = "{" Then
        closing = "}"
    ElseIf opening = "[" Then
        closing = "]"
    Else
        Err.Raise vbObjectError + 7436, , "Expected JSON object/array."
    End If
    For i = startPos To Len(src)
        c = Mid$(src, i, 1)
        If inString Then
            If escaped Then
                escaped = False
            ElseIf c = "\" Then
                escaped = True
            ElseIf c = """" Then
                inString = False
            End If
        ElseIf c = """" Then
            inString = True
        ElseIf c = opening Then
            depth = depth + 1
        ElseIf c = closing Then
            depth = depth - 1
            If depth = 0 Then
                TSA_JSONMatchingEnd = i
                Exit Function
            End If
        End If
    Next i
    Err.Raise vbObjectError + 7437, , "Unterminated JSON object/array."
End Function

Private Function TSA_RawRowsToJSON(ByVal rows As Collection) As String
    Dim i As Long, result As String
    result = "["
    For i = 1 To rows.Count
        If i > 1 Then result = result & ","
        result = result & CStr(rows(i))
    Next i
    TSA_RawRowsToJSON = result & "]"
End Function

' =============================================================
' PALETTE ENTRY POINT: customise resolver and connection constants.
' Use a single AutoLISP palette command TSAOVLAPP, provided alongside.
' =============================================================

' Roadway ID is held in the AutoLISP variable secrid after XSTART.
' LISP cannot be read directly as a VBA global, so TSAOVLAPP temporarily
' passes it via AutoCAD's numeric user variable USERR1. The LISP command
' restores the previous USERR1 value after VBA returns.
' Run TSAOVLAPP, not the parameterless VBA macro by itself.
Public Function TSA_ResolveSectionRdID() As Long
    Dim rawValue As Variant, n As Double
    On Error GoTo InvalidRoadway
    rawValue = ThisDrawing.GetVariable("USERR1")
    If Not IsNumeric(rawValue) Then GoTo InvalidRoadway
    n = CDbl(rawValue)
    If n <= 0 Or n > 2147483647# Then GoTo InvalidRoadway
    If n <> Fix(n) Then GoTo InvalidRoadway
    TSA_ResolveSectionRdID = CLng(n)
    Exit Function
InvalidRoadway:
    TSA_ResolveSectionRdID = 0
End Function

' Parameterless macro suitable for LISP vl-vbarun and palette.
Public Sub TSA_ExportSectionDataToApp()
    Dim rdID As Long
    rdID = TSA_ResolveSectionRdID()
    If rdID <= 0 Then
        MsgBox "Cannot determine RdID for current section." & vbCrLf & _
               "Run XSTART to populate secrid, then invoke TSAOVLAPP from the tool palette.", vbExclamation
        Exit Sub
    End If
    TSA_ExportSectionDataToAppForRdID rdID
End Sub

' Testable API accepting an externally obtained RdID.
Public Sub TSA_ExportSectionDataToAppForRdID(ByVal rdID As Long)
    On Error GoTo Failure
    Dim jsonPath As String, fallbackPath As String
    Dim accessToken As String
    If rdID <= 0 Then Err.Raise vbObjectError + 7440, , "Invalid RdID."
    If InStr(1, TSA_APP_SUPABASE_URL, "YOUR-PROJECT", vbTextCompare) > 0 Or _
       InStr(1, TSA_APP_PUBLISHABLE_KEY, "YOUR-PUBLISHABLE-KEY", vbTextCompare) > 0 Then _
       Err.Raise vbObjectError + 7441, , "Configure the Supabase URL and publishable key in module constants."
    jsonPath = TSA_APP_OUTPUT_FOLDER & "\TSA_section_overlay.json"
    fallbackPath = TSA_APP_OUTPUT_FOLDER & "\tsa_overlay_fallback.txt"
    If Dir$(fallbackPath) = "" Then Err.Raise vbObjectError + 7442, , _
        "AutoLISP fallback file missing; run TSAOVLAPP from the tool palette."
    ' Token code lives in your existing authenticated VBA project.
    ' If its public variable is named SupabaseAccessToken, use the next two lines.
    EnsureSupabaseToken
    accessToken = SupabaseAccessToken
    If Len(Trim$(accessToken)) = 0 Then Err.Raise vbObjectError + 7443, , "Supabase session unavailable."
    TSA_ExportCADOverlayJSONWithFallback rdID, jsonPath, fallbackPath
    TSA_UploadCADOverlayJSON jsonPath, TSA_APP_SUPABASE_URL, _
                             TSA_APP_PUBLISHABLE_KEY, accessToken, rdID
    Exit Sub
Failure:
    MsgBox "Export Section Data to App failed:" & vbCrLf & Err.Description, vbCritical
End Sub
