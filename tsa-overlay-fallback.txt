;;; TSA generic CAD overlay fallback for problematic objects.
;;; Run TSAOVLFALLBACK while the same mining-section UCS is active as in VBA.
;;; Writes section-space coordinates for polylines and MLeader text locations.
;;; Lines: P|HANDLE|x,y;x,y;...   M|HANDLE|x,y
(vl-load-com)
(defun tsaovl:num (x) (vl-string-translate "," "." (rtos x 2 12)))
(defun tsaovl:xy (p / q)
  (setq q (trans p 0 1))
  (strcat (tsaovl:num (car q)) "," (tsaovl:num (cadr q))))
(defun tsaovl:join (items sep / res)
  (setq res "")
  (foreach a items (setq res (strcat res (if (= res "") "" sep) a))) res)
(defun tsaovl:polypts (obj / endp param p pts)
  ;; vlax-curve-getPointAtParam returns WCS points even for OCS polylines.
  (setq endp (vl-catch-all-apply 'vlax-curve-getEndParam (list obj)))
  (if (vl-catch-all-error-p endp) nil
    (progn
      (setq param 0.0 pts nil)
      (while (<= param (+ endp 0.0000001))
        (setq p (vl-catch-all-apply 'vlax-curve-getPointAtParam (list obj param)))
        (if (not (vl-catch-all-error-p p))
          (setq pts (cons (tsaovl:xy p) pts)))
        (setq param (+ param 1.0)))
      (reverse pts))))
(defun tsaovl:mltext (obj / p v)
  ;; TextLocation is exposed by some AutoCAD ActiveX releases.
  ;; If absent, don't guess an insertion point from a WCS bounding box.
  (setq v (vl-catch-all-apply 'vlax-get-property (list obj 'TextLocation)))
  (if (vl-catch-all-error-p v) nil
    (progn
      (cond
        ((listp v) v)
        ((= (type v) 'VARIANT)
          (setq p (vl-catch-all-apply 'vlax-safearray->list (list (vlax-variant-value v))))
          (if (vl-catch-all-error-p p) nil p))
        ((= (type v) 'SAFEARRAY)
          (setq p (vl-catch-all-apply 'vlax-safearray->list (list v)))
          (if (vl-catch-all-error-p p) nil p))
        (T nil)))))
(defun c:TSAOVLFALLBACK (/ fname f ss i en obj typ h pts p nP nM nSkip)
  (setq fname (if (and (boundp 'tsaovl:*output-path*) tsaovl:*output-path*)
                  tsaovl:*output-path*
                  (getfiled "TSA fallback geometry output" "C:\Temp\tsa_overlay_fallback.txt" "txt" 1)))
  (if fname
    (progn
      (setq f (open fname "w") nP 0 nM 0 nSkip 0)
      (if (null f) (prompt "\nCannot open fallback output file.")
        (progn
          (setq ss (ssget "_X" '((0 . "LWPOLYLINE,POLYLINE,MULTILEADER") (410 . "Model"))))
          (if ss
            (progn
              (setq i 0)
              (while (< i (sslength ss))
                (setq en (ssname ss i) obj (vlax-ename->vla-object en)
                      typ (cdr (assoc 0 (entget en)))
                      h (vla-get-Handle obj))
                (if (member (strcase (vla-get-Layer obj)) '("PLANNING" "XOVERSXSECT" "XOVERSOFFSECTION"))
                  (cond
                    ((member typ '("LWPOLYLINE" "POLYLINE"))
                      (setq pts (tsaovl:polypts obj))
                      (if (> (length pts) 1)
                        (progn (write-line (strcat "P|" h "|" (tsaovl:join pts ";")) f)
                          (setq nP (1+ nP)))
                        (setq nSkip (1+ nSkip))))
                    ((= typ "MULTILEADER")
                      (setq p (tsaovl:mltext obj))
                      (if (and (listp p) (>= (length p) 3))
                        (progn (write-line (strcat "M|" h "|" (tsaovl:xy p)) f)
                          (setq nM (1+ nM)))
                        (setq nSkip (1+ nSkip))))))
                (setq i (1+ i)))))
          (close f)
          (prompt (strcat "\nTSA fallback: " (itoa nP) " polylines, "
                          (itoa nM) " leader text positions; " (itoa nSkip) " unavailable."))
          (prompt (strcat "\nWrote: " fname))))))
  (princ))
(princ "\nTSA overlay fallback loaded. Run TSAOVLFALLBACK with section UCS active.")
(princ)

;;; Stage 1.7 DXF-context MLeader geometry extraction.
;;; DXF group numbers repeat in nested sections. Only consume 12 inside
;;; CONTEXT_DATA, 110 before LEADER{, and 10 inside the LEADER{ section.
;;; WCS coordinates are transformed to the *active* section UCS by tsaovl:xy.
(defun tsaovl:mldxf (en / data item code val context leader line textpt arrow landing)
  (setq data (entget en) context nil leader nil line nil
        textpt nil arrow nil landing nil)
  (foreach item data
    (setq code (car item) val (cdr item))
    (cond
      ((and (= code 300) (= val "CONTEXT_DATA{"))
       (setq context T leader nil line nil))
      ((and context (= code 302) (= val "LEADER{"))
       (setq leader T line nil))
      ((and context leader (= code 304) (= val "LEADER_LINE{"))
       (setq line T))
      ((and context leader line (= code 305) (= val "}"))
       (setq line nil))
      ((and context leader (not line) (= code 303) (= val "}"))
       (setq leader nil))
      ((and context (not leader) (= code 301) (= val "}"))
       (setq context nil))
      ((and context (not leader) (= code 12) (listp val)
            (= (length val) 3) (null textpt))
       (setq textpt val))
      ((and context (not leader) (= code 110) (listp val)
            (= (length val) 3) (null arrow))
       (setq arrow val))
      ((and context leader (not line) (= code 10) (listp val)
            (= (length val) 3) (null landing))
       (setq landing val))))
  (list textpt arrow landing))

;;; Override the Stage 1.5 command, retain all existing polyline behaviour.
;;; M|handle|text-x,text-y  L|handle|arrow-x,arrow-y;landing-x,landing-y
(defun c:TSAOVLFALLBACK (/ fname f ss i en obj typ h pts ml nP nM nL nSkip)
  (setq fname (if (and (boundp 'tsaovl:*output-path*) tsaovl:*output-path*)
                  tsaovl:*output-path*
                  (getfiled "TSA fallback geometry output" "C:\Temp\tsa_overlay_fallback.txt" "txt" 1)))
  (if fname
    (progn
      (setq f (open fname "w") nP 0 nM 0 nL 0 nSkip 0)
      (if (null f) (prompt "\nCannot open fallback output file.")
        (progn
          (setq ss (ssget "_X" '((0 . "LWPOLYLINE,POLYLINE,MULTILEADER") (410 . "Model"))))
          (if ss
            (progn (setq i 0)
              (while (< i (sslength ss))
                (setq en (ssname ss i)
                      obj (vlax-ename->vla-object en)
                      typ (cdr (assoc 0 (entget en)))
                      h (vla-get-Handle obj))
                (if (member (strcase (vla-get-Layer obj))
                            '("PLANNING" "XOVERSXSECT" "XOVERSOFFSECTION"))
                  (cond
                    ((member typ '("LWPOLYLINE" "POLYLINE"))
                     (setq pts (tsaovl:polypts obj))
                     (if (> (length pts) 1)
                       (progn (write-line (strcat "P|" h "|" (tsaovl:join pts ";")) f)
                              (setq nP (1+ nP)))
                       (setq nSkip (1+ nSkip))))
                    ((= typ "MULTILEADER")
                     (setq ml (tsaovl:mldxf en))
                     (if (car ml)
                       (progn (write-line (strcat "M|" h "|" (tsaovl:xy (car ml))) f)
                              (setq nM (1+ nM)))
                       (setq nSkip (1+ nSkip)))
                     (if (and (cadr ml) (caddr ml))
                       (progn
                         (write-line (strcat "L|" h "|" (tsaovl:xy (cadr ml)) ";"
                                             (tsaovl:xy (caddr ml))) f)
                         (setq nL (1+ nL)))
                       (setq nSkip (1+ nSkip))))))
                (setq i (1+ i)))))
          (close f)
          (prompt (strcat "\nTSA fallback: " (itoa nP) " polylines, "
                          (itoa nM) " MLeader text anchors, "
                          (itoa nL) " leader lines; " (itoa nSkip) " unavailable."))
          (prompt (strcat "\nWrote: " fname))))))
  (princ))
(princ "\nStage 1.7 TSA overlay fallback loaded. Run TSAOVLFALLBACK.")
(princ)

;;; Tool palette entry: ^P^C^(geold 8);(XSTART);(TSAOVLAPP)
;;; XSTART sets the AutoLISP variable secrid. Pass its value temporarily via
;;; USERR1 because vl-vbarun cannot supply VBA procedure arguments.
;;; Never run XSTART automatically here: it may perform other drawing changes.
(defun TSAOVLAPP (/ tsaovl:*output-path* previous-id rid result)
  (cond
    ((not (and (boundp 'secrid)
               (numberp secrid)
               (> secrid 0)
               (= secrid (fix secrid))
               (<= secrid 2147483647)))
     (prompt "\nNo valid secrid. Run XSTART for this drawing before exporting."))
    (T
     (setq rid (fix secrid))
     (setq tsaovl:*output-path* "C:\\Temp\\tsa_overlay_fallback.txt")
     (if (findfile tsaovl:*output-path*) (vl-file-delete tsaovl:*output-path*))
     (c:TSAOVLFALLBACK)
     (if (findfile tsaovl:*output-path*)
       (progn
         (setq previous-id (getvar "USERR1"))
         (setq result
           (vl-catch-all-apply
             '(lambda ()
                (setvar "USERR1" (float rid))
                (vl-vbarun "TSA_ExportSectionDataToApp"))))
         (setvar "USERR1" previous-id)
         (if (vl-catch-all-error-p result)
           (prompt (strcat "\nTSA VBA launch failed: " (vl-catch-all-error-message result)))))
       (prompt "\nNo fallback file produced; export/upload cancelled."))))
  (princ))
(princ "\nTool palette entry: TSAOVLAPP (run XSTART first; load VBA project).")
(princ)
