Attribute VB_Name = "modTSASectionCADOverlayUpload"
Option Explicit

' Stage 2 - Upload a complete Stage 1.7 CAD overlay JSON snapshot to Supabase.
' Depends on JsonConverter.bas and TSA_CADExportLayers() from Stage 1.7.
' The expectedRdID is mandatory and protects against an incorrect export argument.
' Upserts one row per (RdID, layerName); empty arrays clear an existing layer.
' No deletes, no edits to GEO_SectionFeatures.
'
' Example after EnsureSupabaseToken:
' TSA_UploadCADOverlayJSON "C:\Temp\TSA_section_overlay.json", _
'     "https://YOUR-PROJECT.supabase.co", "YOUR-PUBLISHABLE-KEY", _
'     SupabaseAccessToken, 8729
'
Public Sub TSA_UploadCADOverlayJSON( _
    ByVal filePath As String, ByVal supabaseUrl As String, _
    ByVal publishableKey As String, ByVal accessToken As String, _
    ByVal expectedRdID As Long)

    On Error GoTo UploadFailure
    Dim stm As Object, root As Object, section As Object
    Dim layers As Collection, layer As Variant, entities As Collection
    Dim rows As Collection, payload As Object, seen As Object
    Dim cfg As Variant, conf As Variant, name As String
    Dim rd As Long, drawing As String, rev As String
    Dim warnCount As Long, raw As String, endpoint As String
    Dim request As Object, result As String

    If expectedRdID <= 0 Then Err.Raise vbObjectError + 7400, , "Expected RdID must be positive."
    If Len(Trim$(publishableKey)) = 0 Or Len(Trim$(accessToken)) = 0 Then _
        Err.Raise vbObjectError + 7401, , "Logged-in user's access token and publishable key are required."
    If Len(Trim$(supabaseUrl)) = 0 Then Err.Raise vbObjectError + 7402, , "Supabase project URL required."

    Set stm = CreateObject("ADODB.Stream")
    stm.Type = 2
    stm.Charset = "utf-8"
    stm.Open
    stm.LoadFromFile filePath
    raw = stm.ReadText(-1)
    stm.Close
    Set stm = Nothing
    If Len(Trim$(raw)) = 0 Then Err.Raise vbObjectError + 7403, , "Overlay JSON file is empty."
    ' ADODB.Stream normally consumes UTF-8 BOM, but guard against retained BOM.
    If Left$(raw, 1) = ChrW$(&HFEFF) Then raw = Mid$(raw, 2)

    Set root = JsonConverter.ParseJson(raw)
    If CStr(root("source")) <> "CAD_OVERLAY" Then Err.Raise vbObjectError + 7404, , "Unexpected JSON source."
    If CLng(root("formatVersion")) <> 1 Then Err.Raise vbObjectError + 7405, , "Unsupported formatVersion."
    Set section = root("section")
    rd = CLng(section("RdID"))
    If rd <> expectedRdID Then Err.Raise vbObjectError + 7406, , _
        "Roadway ID mismatch. File RdID=" & rd & "; expected=" & expectedRdID & ". Regenerate export with the correct RdID."
    If CStr(section("coordinateSystem")) <> "CHAINAGE_ELEVATION" Then _
        Err.Raise vbObjectError + 7407, , "Expected CHAINAGE_ELEVATION coordinates."
    If CStr(section("units")) <> "metres" Then Err.Raise vbObjectError + 7408, , "Expected metres."
    drawing = Trim$(CStr(root("sourceDrawing")))
    rev = Trim$(CStr(root("sourceRevision")))
    If Len(drawing) = 0 Or Len(rev) = 0 Then Err.Raise vbObjectError + 7409, , "Drawing/revision missing."

    Set layers = root("layers")
    Set seen = CreateObject("Scripting.Dictionary")
    seen.CompareMode = vbTextCompare
    Set rows = New Collection
    conf = TSA_CADExportLayers()

    For Each layer In layers
        name = CStr(layer("layerName"))
        If seen.Exists(name) Then Err.Raise vbObjectError + 7410, , "Duplicate layer in export: " & name
        Dim configured As Boolean
        configured = False
        For Each cfg In conf
            If StrComp(name, CStr(cfg), vbTextCompare) = 0 Then configured = True
        Next cfg
        If Not configured Then Err.Raise vbObjectError + 7411, , "Unexpected layer: " & name
        If Not CBool(layer("layerExists")) Then _
            Err.Raise vbObjectError + 7412, , "Configured drawing layer missing: " & name
        Set entities = layer("entities")
        Set payload = CreateObject("Scripting.Dictionary")
        payload.Add "RdID", rd
        payload.Add "layerName", name
        payload.Add "entities", entities
        payload.Add "sourceDrawing", drawing
        payload.Add "sourceRevision", rev
        rows.Add payload
        seen.Add name, True
    Next layer
    For Each cfg In conf
        If Not seen.Exists(CStr(cfg)) Then Err.Raise vbObjectError + 7413, , "Missing layer: " & CStr(cfg)
    Next cfg
    If rows.Count = 0 Then Err.Raise vbObjectError + 7414, , "No layers to upload."

    warnCount = root("warnings").Count
    Debug.Print "Overlay upload RdID=" & rd & ", layers=" & rows.Count & _
                ", extraction notices=" & warnCount
    If warnCount > 0 Then Debug.Print "Review local JSON warnings before using overlay operationally."

    If Right$(supabaseUrl, 1) = "/" Then supabaseUrl = Left$(supabaseUrl, Len(supabaseUrl) - 1)
    endpoint = supabaseUrl & "/rest/v1/GEO_SectionCADOverlays?on_conflict=RdID%2ClayerName"
    Set request = CreateObject("MSXML2.ServerXMLHTTP.6.0")
    request.setTimeouts 15000, 15000, 30000, 120000
    request.Open "POST", endpoint, False
    request.setRequestHeader "apikey", publishableKey
    request.setRequestHeader "Authorization", "Bearer " & accessToken
    request.setRequestHeader "Content-Type", "application/json; charset=utf-8"
    request.setRequestHeader "Prefer", "resolution=merge-duplicates,return=minimal"
    request.send JsonConverter.ConvertToJson(rows)
    result = request.responseText
    Debug.Print "Overlay upload HTTP status: " & request.Status
    If request.Status < 200 Or request.Status >= 300 Then _
        Err.Raise vbObjectError + 7415, , "Supabase HTTP " & request.Status & ": " & Left$(result, 1800)

    MsgBox "CAD overlay snapshots uploaded successfully." & vbCrLf & _
           "Roadway: " & rd & vbCrLf & "Layers: " & rows.Count & vbCrLf & _
           "Local extraction notices: " & warnCount, vbInformation
    Exit Sub

UploadFailure:
    Dim errorText As String
    errorText = Err.Description
    On Error Resume Next
    If Not stm Is Nothing Then stm.Close
    On Error GoTo 0
    MsgBox "CAD overlay upload failed:" & vbCrLf & errorText, vbCritical
End Sub
