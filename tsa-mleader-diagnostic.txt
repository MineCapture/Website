;;; TSA Stage 1.6 - Read-only MLeader diagnostic
;;; Does not change entities, drawing UCS, layer state or the VBA export.
;;; Command: TSA_MLEADER_DIAG
;;; Requires Visual LISP support (vl-load-com) and an active section UCS.
(vl-load-com)

(defun tsa16:write (f s) (write-line s f))
(defun tsa16:fmt (v) (vl-prin1-to-string v))
(defun tsa16:err (v) (if (vl-catch-all-error-p v) (vl-catch-all-error-message v) nil))
(defun tsa16:tryprop (obj name / v)
  (setq v (vl-catch-all-apply 'vlax-get-property (list obj name)))
  (if (tsa16:err v) (strcat "UNAVAILABLE: " (tsa16:err v)) (tsa16:fmt v)))
(defun tsa16:unpack (v / t1)
  (cond ((= (type v) 'VARIANT) (tsa16:unpack (vlax-variant-value v)))
        ((= (type v) 'SAFEARRAY)
         (setq t1 (vl-catch-all-apply 'vlax-safearray->list (list v)))
         (if (tsa16:err t1) nil t1))
        ((listp v) v)
        (T nil)))
(defun tsa16:point (p / q)
  (setq p (tsa16:unpack p))
  (if (and p (>= (length p) 3)
           (numberp (car p)) (numberp (cadr p)) (numberp (caddr p)))
    (progn
      (setq q (vl-catch-all-apply 'trans (list p 0 1)))
      (if (tsa16:err q) (strcat "UCS conversion error: " (tsa16:err q))
        (strcat "WCS=" (tsa16:fmt p) " | UCS=" (tsa16:fmt q))))
    "Not a 3D point"))
(defun tsa16:property (f obj name / v)
  (setq v (vl-catch-all-apply 'vlax-get-property (list obj name)))
  (tsa16:write f (strcat "  PROPERTY " (vl-princ-to-string name) " = "
    (if (tsa16:err v) (strcat "UNAVAILABLE: " (tsa16:err v)) (tsa16:fmt v))))
  (if (and (not (tsa16:err v)) (member name '(TextLocation LandingGap)))
    (tsa16:write f (strcat "    section interpretation: " (tsa16:point v))))
  v)
(defun tsa16:method (f obj name args / v)
  (setq v (vl-catch-all-apply 'vlax-invoke-method (append (list obj name) args)))
  (tsa16:write f (strcat "  METHOD " (vl-princ-to-string name) "(" (tsa16:fmt args) ") = "
    (if (tsa16:err v) (strcat "ERROR: " (tsa16:err v)) (tsa16:fmt v))))
  v)
(defun tsa16:dump-dxf (f en / dat pair code)
  (tsa16:write f "\nDXF (entget with application XDATA):")
  (setq dat (vl-catch-all-apply 'entget (list en (list "*"))))
  (if (tsa16:err dat)
    (tsa16:write f (strcat "  ERROR: " (tsa16:err dat)))
    (foreach pair dat
      (setq code (car pair))
      (tsa16:write f (strcat "  " (itoa code) " : " (tsa16:fmt (cdr pair))))
      (if (and (member code '(10 11 12 13 110 111 112 210))
               (listp (cdr pair)) (= (length (cdr pair)) 3)
               (numberp (cadr pair)))
        (tsa16:write f (strcat "    NOTE: raw DXF coordinate; its OCS/WCS basis varies by code/entity. "
                                "Do not treat as WCS without checking."))))))
(defun tsa16:inspect (f handle / en obj d typ props count i inds idxs verts child err)
  (tsa16:write f (strcat "\n==================================================\nHANDLE " handle))
  (setq en (handent handle))
  (if (null en) (tsa16:write f "NOT FOUND in current drawing")
    (progn
      (setq d (entget en) typ (cdr (assoc 0 d)))
      (tsa16:write f (strcat "DXF entity type = " (tsa16:fmt typ)))
      (tsa16:write f (strcat "Layer = " (tsa16:fmt (cdr (assoc 8 d)))))
      (if (/= typ "MULTILEADER")
        (tsa16:write f "WARNING: this handle is not a MULTILEADER.")
        (progn
          (setq obj (vl-catch-all-apply 'vlax-ename->vla-object (list en)))
          (if (tsa16:err obj) (tsa16:write f (strcat "Cannot access ActiveX: " (tsa16:err obj)))
            (progn
              (tsa16:write f "\nActiveX PROPERTIES (availability depends on AutoCAD version):")
              (foreach prop '(ObjectName Handle Layer ContentType TextString TextLocation
                               TextHeight TextRotation TextWidth TextLeftAttachmentType
                               TextRightAttachmentType TextTopAttachmentType
                               TextBottomAttachmentType TextAttachmentDirection
                               TextJustify TextDirection LeaderCount LeaderLineColor
                               LeaderLineType LeaderLineWeight LandingGap DoglegLength
                               EnableLanding EnableDogleg ArrowheadSize StyleName ScaleFactor)
                (tsa16:property f obj prop))
              (tsa16:write f "\nActiveX leader-line method probes:")
              ;; Cluster identifiers are not assumed to be valid integer sequence.
              (setq count (vl-catch-all-apply 'vlax-get-property (list obj 'LeaderCount)))
              (if (and (not (tsa16:err count)) (numberp count))
                (progn
                  (tsa16:write f (strcat "LeaderCount=" (itoa count)))
                  (setq i 0)
                  (while (and (< i count) (< i 20))
                    (setq inds (tsa16:method f obj 'GetLeaderLineIndexes (list i)))
                    (setq idxs (tsa16:unpack inds))
                    (if (and idxs (not (tsa16:err inds)))
                      (foreach child idxs
                        (if (numberp child)
                          (progn
                            (setq verts (tsa16:method f obj 'GetLeaderLineVertices (list child)))
                            (tsa16:write f (strcat "    vertex raw data = " (tsa16:fmt (tsa16:unpack verts))))))))
                    (setq i (1+ i)))))
              ;; Direct line zero is useful where no clusters are reachable.
              (tsa16:method f obj 'GetLeaderLineVertices (list 0))))))
      (tsa16:dump-dxf f en))))
(defun c:TSA_MLEADER_DIAG (/ fname f handle ent ed)
  (setq fname (getfiled "Save TSA MLeader diagnostic" "C:\\Temp\\tsa_mleader_diagnostic.txt" "txt" 1))
  (if (null fname) (prompt "\nCancelled.")
    (progn
      (setq f (open fname "w"))
      (if (null f) (prompt "\nUnable to open output file.")
        (progn
          (tsa16:write f "TSA MLeader diagnostic - read only")
          (tsa16:write f (strcat "Drawing: " (getvar "DWGNAME")))
          (tsa16:write f (strcat "UCSNAME: " (tsa16:fmt (getvar "UCSNAME"))))
          (tsa16:write f (strcat "UCSORG: " (tsa16:fmt (getvar "UCSORG"))))
          (tsa16:write f (strcat "UCSXDIR: " (tsa16:fmt (getvar "UCSXDIR"))))
          (tsa16:write f (strcat "UCSYDIR: " (tsa16:fmt (getvar "UCSYDIR"))))
          (foreach handle '("5C5D3" "74DBB" "5C5CF" "5C5D2")
            (tsa16:inspect f handle))
          (close f)
          (prompt (strcat "\nTSA MLeader diagnostic created: " fname))
          (prompt "\nContains the two missing leaders plus two working reference leaders.")))))
  (princ))
(princ "\nTSA Stage 1.6 MLeader diagnostic loaded. Run TSA_MLEADER_DIAG.")
(princ)
